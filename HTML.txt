  <script>
    // Инициализация Telegram WebApp
    const tg = window.Telegram?.WebApp;
    if (tg) {
      tg.ready();
      tg.expand();
    }

    function switchTab(tabId, el) {
      document.querySelectorAll('.tab-content').forEach(t => t.classList.remove('active'));
      document.querySelectorAll('.nav-item').forEach(n => n.classList.remove('active'));
      
      document.getElementById('tab-' + tabId).classList.add('active');
      el.classList.add('active');
      
      if (tg?.HapticFeedback) tg.HapticFeedback.impactOccurred('light');
    }

    function toggleModal(show) {
      document.getElementById('settingsModal').classList.toggle('active', show);
      if (tg?.HapticFeedback) tg.HapticFeedback.impactOccurred('medium');
    }

    function setPill(el) {
      el.parentElement.querySelectorAll('.pill').forEach(p => p.classList.remove('active'));
      el.classList.add('active');
      if (tg?.HapticFeedback) tg.HapticFeedback.selectionChanged();
    }

    function addTrade(isWin) {
      alert(isWin ? "✅ Сделка (+ПЛЮС) добавлена в статистику!" : "❌ Сделка (-МИНУС) добавлена в статистику!");
      if (tg?.HapticFeedback) tg.HapticFeedback.notificationOccurred(isWin ? 'success' : 'error');
    }

    function buySub(plan) {
      alert(`🎉 Переход к оплате тарифа: ${plan}`);
    }
  </script>
